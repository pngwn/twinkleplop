// --- generics whose parameter list closes on a coalesced `>` ---

// the tokenizer emits a run of `>` as one right-shift operator, so both
// groups here close on a single token. the walkers that only understood a
// lone `>` never found the close and gave up on the whole declaration:
// the name stayed an identifier, `T` stayed an identifier, and `base` was
// never tagged as a parameter.
interface Introspectable {
  create_enhanced_introspector<T extends Record<string, unknown>>(base: T): T;
}

  enhance<T extends Record<string, unknown>>(base: T): T {
    return base;
  }
}

function enhance<T extends Record<string, unknown>>(base: T): T {
  return base;
}
